1. write a python code to find the factorial of a number
```python
def factorial(n):
    """
    Calculates the factorial of a non-negative integer.
    """
    if n < 0:
        return "Factorial is not defined for negative numbers"
    elif n == 0:
        return 1
    else:
        result = 1
        for i in range(1, n + 1):
            result *= i
        return result

# Example usage:
# num = 5
# print(f"The factorial of {num} is {factorial(num)}") # Output: The factorial of 5 is 120
```

2. write a java code to find the prime numbers between 1 to 100
```java
public class PrimeNumbers {
    public static void main(String[] args) {
        System.out.println("Prime numbers between 1 and 100 are:");
        for (int i = 2; i <= 100; i++) {
            boolean isPrime = true;
            // Check for divisibility from 2 up to the square root of i
            for (int j = 2; j <= Math.sqrt(i); j++) {
                if (i % j == 0) {
                    isPrime = false;
                    break; // Not a prime number, no need to check further
                }
            }
            if (isPrime) {
                System.out.print(i + " ");
            }
        }
        System.out.println(); // For a new line after printing all primes
    }
}
```

3. write a c++ code to find the Armstrong numbers between 1 to 1000
```cpp
#include <iostream>
#include <cmath> // Required for std::pow

int main() {
    std::cout << "Armstrong numbers between 1 and 1000 are:" << std::endl;
    for (int i = 1; i <= 1000; ++i) {
        int originalNumber = i;
        int sumOfPowers = 0;
        int numDigits = 0;
        int temp = originalNumber;

        // Count the number of digits
        if (originalNumber == 0) {
            numDigits = 1;
        } else {
            while (temp > 0) {
                temp /= 10;
                numDigits++;
            }
        }

        temp = originalNumber; // Reset temp for digit extraction and sum calculation

        // Calculate the sum of digits raised to the power of numDigits
        while (temp > 0) {
            int digit = temp % 10;
            sumOfPowers += static_cast<int>(std::pow(digit, numDigits));
            temp /= 10;
        }

        if (sumOfPowers == originalNumber) {
            std::cout << originalNumber << " ";
        }
    }
    std::cout << std::endl;
    return 0;
}
```

4. write a javascript code to find the sum of the first 20 even numbers
```javascript
function sumOfFirstNEvenNumbers(n) {
    let sum = 0;
    // Iterate from 1 to n to get the first n even numbers
    for (let i = 1; i <= n; i++) {
        sum += (2 * i); // The i-th even number is 2*i
    }
    return sum;
}
```
